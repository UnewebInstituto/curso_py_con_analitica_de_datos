#-----------------------------------------------------------------

import csv
import statistics

# Listas para almacenar los valores de cada columna
ages = []
experiences = []
salaries = []
# Ruta del archivo CSV
archivo_csv = "./SESIÓN_1/Salary_Data_Test.csv"

# Abrir el archivo CSV manejando posibles marcas de orden de bytes (BOM)
with open(archivo_csv, mode='r', encoding='utf-8', errors='ignore') as archivo:
  content = archivo.read().lstrip('\ufeff')
  lines = content.splitlines()
  lector = csv.reader(lines)

  # Leer la cabecera para identificar los índices de las columnas
  header = next(lector)
  age_idx = header.index('Age')
  exp_idx = header.index('Years of Experience')
  sal_idx = header.index('Salary')

  # Recorrer el archivo fila por fila y extraer los datos numéricos
  for fila in lector:
    if len(fila) <= max(age_idx, exp_idx, sal_idx):
      continue
    try:
      if fila[age_idx].strip():
        ages.append(float(fila[age_idx]))
      if fila[exp_idx].strip():
        experiences.append(float(fila[exp_idx]))
      if fila[sal_idx].strip():
        salaries.append(float(fila[sal_idx]))
    except ValueError:
      continue  # Salta filas que tengan errores de conversión


# Función auxiliar para calcular las métricas estadísticas requeridas
def calcular_metricas(datos):
  n = len(datos)
  media = sum(datos) / n
  mediana = statistics.median(datos)
  minimo = min(datos)
  maximo = max(datos)
  desv_std = statistics.stdev(datos) if n > 1 else 0
  return {
      'media': media,
      'mediana': mediana,
      'min': minimo,
      'max': maximo,
      'std': desv_std,
  }


# Obtener resultados para cada variable
stats_age = calcular_metricas(ages)
stats_exp = calcular_metricas(experiences)
stats_sal = calcular_metricas(salaries)

# Imprimir resultados formateados
print('--- EDAD (Age) ---')
print(
    f"Media: {stats_age['media']:.2f} | Mediana: {stats_age['mediana']:.2f}"
    f" | Mín: {stats_age['min']:.2f} | Máx: {stats_age['max']:.2f} | Desv.Std:"
    f" {stats_age['std']:.2f}"
)

print('\n--- AÑOS DE EXPERIENCIA (Years of Experience) ---')
print(
    f"Media: {stats_exp['media']:.2f} | Mediana: {stats_exp['mediana']:.2f}"
    f" | Mín: {stats_exp['min']:.2f} | Máx: {stats_exp['max']:.2f} | Desv.Std:"
    f" {stats_exp['std']:.2f}"
)

print('\n--- SALARIO (Salary) ---')
print(
    f"Media: ${stats_sal['media']:,.2f} | Mediana:"
    f" ${stats_sal['mediana']:,.2f} | Mín: ${stats_sal['min']:,.2f} | Máx:"
    f" ${stats_sal['max']:,.2f} | Desv.Std: ${stats_sal['std']:,.2f}"
)

Para realizar el mismo análisis y procesamiento estadístico utilizando datos provenientes directamente de una base de datos PostgreSQL (en lugar de leerlos de un archivo CSV plano), la estrategia cambia: nos conectamos a la base de datos, extraemos los registros mediante una consulta SQL, los cargamos en estructuras nativas de Python y aplicamos exactamente las mismas funciones estadísticas que ya construimos.

Para definir una tabla en una base de datos PostgreSQL, se utiliza la sentencia SQL CREATE TABLE. Esta instrucción permite especificar el nombre de la tabla, sus columnas, los tipos de datos que almacenará cada una y las restricciones o reglas de integridad (como claves primarias, valores obligatorios o únicos).

Tomando como referencia los datos que hemos estado analizando en nuestro curso (el dataset de salarios y perfiles profesionales), un ejemplo práctico de definición de tabla en PostgreSQL sería el siguiente:

(SE USARÁ PGADMIN PARA ESTA DEFINICIÓN)

CREATE TABLE salary_data (
    id SERIAL PRIMARY KEY,                          -- Identificador único por registro
    age INTEGER,
    gender CHARACTER VARYING(40),
    education_level TEXT,
    job_title TEXT,
    years_of_experience NUMERIC(5,2),
    salary NUMERIC(13,2)
);


Dado que estamos limitados al uso de la librería estándar y módulos nativos, el enfoque ideal en un entorno de IDLE y Python puro es utilizar el módulo psycopg2 (o psycopg según la versión instalada) para la conexión a PostgreSQL.

Aquí tienes el script completo en Python para lograrlo:

import statistics
import psycopg2

# Configuración de la conexión a PostgreSQL
# (Ajusta estos parámetros según los datos de tu servidor local o remoto)
db_config = {
    'dbname': 'db_bm_py_analitica_curso',
    'user': 'postgres',
    'password': '1234567',
    'host': 'localhost',
    'port': '5432',
}

# Listas para almacenar los valores extraídos de la BD
ages = []
experiences = []
salaries = []

try:
  # 1. Conexión a la base de datos PostgreSQL
  conexion = psycopg2.connect(**db_config)
  cursor = conexion.cursor()

  # 2. Consulta SQL para extraer los datos de la tabla salary_data
  consulta = """
        SELECT age, years_of_experience, salary 
        FROM salary_data 
        WHERE age IS NOT NULL 
          AND years_of_experience IS NOT NULL 
          AND salary IS NOT NULL;
    """

  cursor.execute(consulta)
  registros = cursor.fetchall()

  # 3. Recorrer los registros de la base de datos y llenar las listas
  for fila in registros:
    # Como la consulta ya trae el orden exacto, los mapeamos por índice
    age, years_of_experience, salary = fila

    # Conversión y almacenamiento seguro
    try:
      ages.append(float(age))
      experiences.append(float(years_of_experience))
      salaries.append(float(salary))
    except (ValueError, TypeError):
      continue  # Salta filas con datos anómalos

  cursor.close()
  conexion.close()

except Exception as e:
  print(f'Error al conectar o consultar la base de datos: {e}')
  exit()

def calcular_metricas(datos):
  n = len(datos)
  if n == 0:
    return None
  media = sum(datos) / n
  mediana = statistics.median(datos)
  minimo = min(datos)
  maximo = max(datos)
  desv_std = statistics.stdev(datos) if n > 1 else 0
  return {
      'media': media,
      'mediana': mediana,
      'min': minimo,
      'max': maximo,
      'std': desv_std,
  }


# 5. Obtener resultados estadísticos para cada variable
stats_age = calcular_metricas(ages)
stats_exp = calcular_metricas(experiences)
stats_sal = calcular_metricas(salaries)

# 6. Imprimir resultados formateados
if stats_age and stats_exp and stats_sal:
  print('--- EDAD (Age) [Desde PostgreSQL] ---')
  print(
      f"Media: {stats_age['media']:.2f} | Mediana: {stats_age['mediana']:.2f}"
      f" | Mín: {stats_age['min']:.2f} | Máx: {stats_age['max']:.2f} |"

      f" Desv.Std: {stats_exp['std']:.2f}"
  )

  print('\n--- SALARIO (Salary) ---')
  print(
      f"Media: ${stats_sal['media']:,.2f} | Mediana:"

*(2)*
Para efectuar la adecuación de tu script y realizar el procesamiento analítico leyendo directamente desde un archivo de Excel (Salary_Data_Test.xlsx) en lugar de PostgreSQL, no podemos usar módulos nativos de Python, ya que el formato .xlsx es un archivo binario comprimido en ZIP y Python por defecto no incluye una librería para leer libros de Excel en su biblioteca estándar.

Por lo tanto, la manera estándar y profesional de lograrlo en un entorno de desarrollo en Python es utilizando la librería openpyxl.

A continuación tienes el script adaptado para leer los datos desde Excel, limpiar y procesar las listas numéricas, calcular las métricas estadísticas usando el módulo nativo statistics, y generar el archivo Excel de resultados:

import statistics
import openpyxl

# Ruta del archivo de Excel de entrada
archivo_excel = './Salary_Data_Test.xlsx'

# Listas para almacenar los valores extraídos de Excel
ages = []
experiences = []
salaries = []

try:
  # 1. Cargar el libro de trabajo de Excel
  wb = openpyxl.load_workbook(archivo_excel, data_only=True)
  ws = wb.active  # Selecciona la hoja activa

  # 2. Leer la cabecera (primera fila) para identificar dinámicamente los índices de las columnas
  header = [cell.value for cell in ws[1]]
  age_idx = header.index('Age')
  exp_idx = header.index('Years of Experience')
  sal_idx = header.index('Salary')

  # 3. Recorrer las filas de Excel omitiendo la cabecera (comenzando desde la fila 2)
  for row in ws.iter_rows(min_row=2, values_only=True):
    # Validar que la fila tenga suficientes columnas
    if row is None or len(row) <= max(age_idx, exp_idx, sal_idx):
      continue

    age = row[age_idx]
    experience = row[exp_idx]
    salary = row[sal_idx]

    # Conversión y almacenamiento seguro (filtrando nulos o vacíos)
    try:
      if age is not None and str(age).strip():
        ages.append(float(age))
      if experience is not None and str(experience).strip():
        experiences.append(float(experience))
      if salary is not None and str(salary).strip():
        salaries.append(float(salary))
    except (ValueError, TypeError):
      continue  # Salta filas con datos anómalos o texto corrupto

except Exception as e:
  print(f'Error al leer el archivo de Excel: {e}')
  exit()


# 4. Función auxiliar para calcular las métricas estadísticas (módulo statistics)

  minimo = min(datos)
  maximo = max(datos)
  desv_std = statistics.stdev(datos) if n > 1 else 0
  return {
      'media': media,
      'mediana': mediana,
      'min': minimo,
      'max': maximo,
      'std': desv_std,
  }

### **1. Inicialización y Configuración del Archivo**

* **`archivo_csv = './Bank Customer Churn Prediction.csv'`**: Define la ruta del archivo de datos a procesar.
* **Variables contadoras**: Establece acumuladores en cero (`total_clientes`, `clientes_retenidos` y `clientes_abandonaron`) para registrar las métricas clave durante el recorrido del archivo.

### **2. Apertura Segura y Manejo de BOM (`try-except`)**

Todo el flujo de procesamiento se encuentra protegido dentro de un bloque de control de excepciones:

* **`open(..., encoding='utf-8', errors='ignore')`**: Abre el archivo de texto plano asegurando la compatibilidad de caracteres e ignorando posibles fallos de codificación.
* **`content.read().lstrip('\ufeff')`**: Lee el contenido completo del archivo y remueve de forma preventiva cualquier marca de orden de bytes (*Byte Order Mark* o BOM) que los editores de texto o herramientas ofimáticas suelen incorporar al inicio, evitando errores al interpretar las cabeceras.
* **`csv.reader(lines)`**: Convierte el contenido dividido por líneas en un lector estructurado de filas y columnas.

### **3. Detección Dinámica de la Columna Objetivo**

* **`header = next(lector)`**: Extrae la primera fila correspondiente a los títulos de las columnas.
* **`churn_idx = header.index('churn')`**: Busca de manera inteligente la posición numérica exacta de la columna `'churn'`, permitiendo que el script funcione correctamente incluso si la estructura o el orden de las columnas cambia en futuras versiones del dataset.

### **4. Recorrido, Validación y Conteo Fila por Fila**

Mediante un bucle `for fila in lector:`, el script procesa cada registro aplicando filtros de seguridad:

* **Validación de longitud**: Comprueba que la fila posea los índices necesarios mediante `len(fila) <= churn_idx` para prevenir errores de tipo *IndexError*.
* **Conversión y Clasificación**: Extrae el valor de la celda, limpia sus espacios en blanco con `.strip()` y lo convierte a un número entero (`int`).
* **Acumulación condicional**:
* Si el valor es `0`, incrementa el contador de **clientes retenidos**.
* Si el valor es `1`, incrementa el contador de **clientes que abandonaron**.


* **Manejo de errores internos (`except ValueError`)**: Si encuentra celdas corruptas, vacías o con textos anómalos, la instrucción `continue` descarta la fila de manera limpia sin interrumpir la ejecución general del programa.

### **5. Cálculo de Porcentajes e Informe de Resultados**

* **Proporciones estadísticas**: Una vez finalizado el recorrido, valida que se hayan encontrado registros válidos (`total_clientes > 0`) y calcula los porcentajes correspondientes de retención y abandono.
* **Presentación formateada en consola**: Imprime un reporte ejecutivo claro y profesional, mostrando las cantidades absolutas con separadores de miles y las tasas porcentuales acotadas a dos decimales de precisión (`.2f`).

import csv

# Ruta del archivo CSV de abandono de clientes
archivo_csv = './Bank Customer Churn Prediction.csv'

# Variables contadoras
total_clientes = 0
clientes_retenidos = 0  # Churn = 0
clientes_abandonaron = 0  # Churn = 1

try:
  # 1. Apertura del archivo CSV manejando posibles marcas de orden de bytes (BOM)
  with open(archivo_csv, mode='r', encoding='utf-8', errors='ignore') as archivo:
    content = archivo.read().lstrip('\ufeff')
    lines = content.splitlines()
    lector = csv.reader(lines)

    # 2. Leer la cabecera para identificar dinámicamente el índice de la columna 'churn'
    header = next(lector)
    churn_idx = header.index('churn')

    # 3. Recorrer el archivo fila por fila
    for fila in lector:
      # Validar que la fila tenga suficientes columnas
      if len(fila) <= churn_idx:
        continue

      try:
        # Extraer el valor de la columna churn y convertirlo a entero (0 o 1)
        valor_churn = int(fila[churn_idx].strip())

        total_clientes += 1

        if valor_churn == 0:
          clientes_retenidos += 1
        elif valor_churn == 1:
          clientes_abandonaron += 1

      except ValueError:
        # Salta filas que tengan errores de conversión o datos corruptos
        continue

  # 4. Cálculo de porcentajes
  if total_clientes > 0:
    pct_retenidos = (clientes_retenidos / total_clientes) * 100
    pct_abandonaron = (clientes_abandonaron / total_clientes) * 100

    # 5. Imprimir resultados formateados
    print('--- ANÁLISIS DE FUGA DE CLIENTES (CHURN) ---')
    print(f'Total de clientes analizados: {total_clientes:,}')
    print(
        f'Clientes Retenidos (0): {clientes_retenidos:,} ({pct_retenidos:.2f}%)'
    )
    print(
        f'Clientes que Abandonaron (1): {clientes_abandonaron:,} ({pct_abandonaron:.2f}%)'
    )
  else:
    print('No se encontraron registros válidos para procesar.')

except Exception as e:
  print(f'Ocurrió un error al procesar el archivo CSV: {e}')
